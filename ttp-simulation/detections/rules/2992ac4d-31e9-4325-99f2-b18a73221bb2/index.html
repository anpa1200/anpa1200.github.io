title: ESXi VM Kill Via ESXCLI

date: 2023-09-04
tags:
    - attack.execution

logsource:
    category: process_creation

detection:
    selection:
        Image|endswith: '/esxcli'

            - 'vm process'
            - 'kill'
    condition: selection
